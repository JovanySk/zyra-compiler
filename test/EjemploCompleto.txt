class Calculadora {
    int resultado;
    
    Calculadora() {
        resultado = 0;
    }
    
    static void main() {
        int x;
        int y;
        int res1;
        int res2;
        
        x = 15;
        y = 7;
        
        res1 = sumar(x, y);
        res2 = multiplicar(x, y);
    }
    
    static int sumar(int a, int b) {
        int suma;
        suma = a + b;
        return suma;
    }
    
    static int multiplicar(int x, int y) {
        int producto;
        producto = x * y;
        return producto;
    }
    
    dynamic void procesar() {
        int a;
        int b;
        int temp;
        
        a = 5;
        b = 10;
        temp = sumar(a, b);
        resultado = multiplicar(temp, 2);
    }
}
