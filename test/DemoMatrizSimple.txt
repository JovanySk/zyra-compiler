class DemoMatrizSimple {
    static void main() {
        int[] fila1;
        int[] fila2;
        int[] fila3;
        int suma;
        
        print("=== MATRIZ 3x3 (SIMPLIFICADA) ===");
        fila1 = new int[3];
        fila2 = new int[3];
        fila3 = new int[3];
        
        print("Ingrese fila 1 (3 numeros):");
        fila1[0] = read();
        fila1[1] = read();
        fila1[2] = read();
        
        print("Ingrese fila 2 (3 numeros):");
        fila2[0] = read();
        fila2[1] = read();
        fila2[2] = read();
        
        print("Ingrese fila 3 (3 numeros):");
        fila3[0] = read();
        fila3[1] = read();
        fila3[2] = read();
        
        print("=== CALCULANDO SUMA ===");
        suma = 0;
        suma = suma + fila1[0] + fila1[1] + fila1[2];
        suma = suma + fila2[0] + fila2[1] + fila2[2];
        suma = suma + fila3[0] + fila3[1] + fila3[2];
        
        print("Suma total de la matriz:");
        print(suma);
        print("==============================");
    }
}
