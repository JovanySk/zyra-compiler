class PruebaBucles {
    static void main() {
        int i;
        int suma;
        int producto;
        
        // For loop
        suma = 0;
        i = 0;
        while (i < 10) {
            suma = suma + i;
            i = i + 1;
        }
        
        // While con condicional interno
        producto = 1;
        i = 1;
        while (i <= 5) {
            if (i > 2) {
                producto = producto * i;
            }
            i = i + 1;
        }
        
        // Bucle anidado
        int j;
        i = 0;
        while (i < 3) {
            j = 0;
            while (j < 3) {
                suma = suma + 1;
                j = j + 1;
            }
            i = i + 1;
        }
    }
}
